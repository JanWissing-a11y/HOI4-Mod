# Removal resets its countdown, so reappointment cannot revive an old scheduled reward.

WFL_staff_refresh_dante_grossy = {
    remove_ideas = WFL_staff_dante_grossy_0_idea
    remove_ideas = WFL_staff_dante_grossy_1_idea
    remove_ideas = WFL_staff_dante_grossy_2_idea
    remove_ideas = WFL_staff_dante_grossy_3_idea
    if = {
        limit = {
            WFL_staff_can_operate = yes
            has_country_flag = WFL_staff_dante_grossy_active
        }
        if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_rank value = 3 compare = equals }
            }
            add_ideas = WFL_staff_dante_grossy_3_idea
        }

        else_if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_rank value = 2 compare = equals }
            }
            add_ideas = WFL_staff_dante_grossy_2_idea
        }

        else_if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_rank value = 1 compare = equals }
            }
            add_ideas = WFL_staff_dante_grossy_1_idea
        }

        else_if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_rank value = 0 compare = equals }
            }
            add_ideas = WFL_staff_dante_grossy_0_idea
        }
    }
}

WFL_staff_prepare_dante_grossy = {
    if = {
        limit = {
            WFL_staff_can_operate = yes
            has_country_flag = WFL_staff_dante_grossy_active
            check_variable = { var = WFL_staff_dante_grossy_rank value = 3 compare = less_than }
        }
        set_variable = { WFL_staff_dante_grossy_progress = 0 }
        set_variable = { WFL_staff_dante_grossy_phase = 1 }
        set_variable_to_random = { var = WFL_staff_dante_grossy_days min = 75 max = 121 integer = yes }
        if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_rank value = 2 compare = equals }
            }
            set_variable = { WFL_staff_dante_grossy_duration = 280 }
        }
        else = { set_variable = { WFL_staff_dante_grossy_duration = 140 } }
    }
    else = { set_variable = { WFL_staff_dante_grossy_phase = 0 } set_variable = { WFL_staff_dante_grossy_days = 0 } }
}

WFL_staff_add_dante_grossy = {
    if = {
        limit = {
            WFL_staff_can_operate = yes
            NOT = { has_country_flag = WFL_staff_dante_grossy_active }
            OR = {
                check_variable = { var = WFL_staff_slot_1 value = 1 compare = equals }
                check_variable = { var = WFL_staff_slot_2 value = 1 compare = equals }
                check_variable = { var = WFL_staff_slot_3 value = 1 compare = equals }
            }
        }
        set_country_flag = WFL_staff_dante_grossy_active
        WFL_staff_refresh_dante_grossy = yes
        if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_rank value = 3 compare = less_than }
            }
            WFL_staff_prepare_dante_grossy = yes
        }
        else = { set_variable = { WFL_staff_dante_grossy_phase = 0 } set_variable = { WFL_staff_dante_grossy_days = 0 } set_variable = { WFL_staff_dante_grossy_progress = 1 } }
    }
}

WFL_staff_remove_dante_grossy = {
    if = {
        limit = {
            has_country_flag = WFL_staff_dante_grossy_active
        }
        clr_country_flag = WFL_staff_dante_grossy_active
        if = {
            limit = {
                has_completed_focus = WFL_staff_appoint_dante_grossy
            }
            uncomplete_national_focus = { focus = WFL_staff_appoint_dante_grossy uncomplete_children = no refund_political_power = no }
        }
    }
    remove_ideas = WFL_staff_dante_grossy_0_idea
    remove_ideas = WFL_staff_dante_grossy_1_idea
    remove_ideas = WFL_staff_dante_grossy_2_idea
    remove_ideas = WFL_staff_dante_grossy_3_idea
    if = {
        limit = {
            has_shine_effect_on_focus = WFL_staff_dante_grossy_1
        }
        deactivate_shine_on_focus = WFL_staff_dante_grossy_1
    }
    if = {
        limit = {
            has_shine_effect_on_focus = WFL_staff_dante_grossy_2
        }
        deactivate_shine_on_focus = WFL_staff_dante_grossy_2
    }
    if = {
        limit = {
            has_shine_effect_on_focus = WFL_staff_dante_grossy_3
        }
        deactivate_shine_on_focus = WFL_staff_dante_grossy_3
    }
    set_variable = { WFL_staff_dante_grossy_phase = 0 }
    set_variable = { WFL_staff_dante_grossy_days = 0 }
    set_variable = { WFL_staff_dante_grossy_duration = 0 }
    set_variable = { WFL_staff_dante_grossy_progress = 0 }
    if = {
        limit = {
            check_variable = { var = WFL_staff_slot_1 value = 1 compare = equals }
        }
        set_variable = { WFL_staff_slot_1 = 0 }
    }
    if = {
        limit = {
            check_variable = { var = WFL_staff_slot_2 value = 1 compare = equals }
        }
        set_variable = { WFL_staff_slot_2 = 0 }
    }
    if = {
        limit = {
            check_variable = { var = WFL_staff_slot_3 value = 1 compare = equals }
        }
        set_variable = { WFL_staff_slot_3 = 0 }
    }
}

WFL_staff_tick_dante_grossy = {
    if = {
        limit = {
            WFL_staff_can_operate = yes
            has_country_flag = WFL_staff_dante_grossy_active
        }
        if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_phase value = 1 compare = equals }
            }
            add_to_variable = { WFL_staff_dante_grossy_days = -1 }
            if = {
                limit = {
                    check_variable = { var = WFL_staff_dante_grossy_days value = 0 compare = less_than_or_equals }
                }
                if = {
                    limit = {
                        check_variable = { var = WFL_staff_dante_grossy_rank value = 0 compare = equals }
                    }
                    set_variable = { WFL_staff_dante_grossy_phase = 2 }
                    set_variable = { WFL_staff_dante_grossy_days = WFL_staff_dante_grossy_duration }
                    set_variable = { WFL_staff_dante_grossy_progress = 0 }
                    activate_shine_on_focus = WFL_staff_dante_grossy_1
                }

                else_if = {
                    limit = {
                        check_variable = { var = WFL_staff_dante_grossy_rank value = 1 compare = equals }
                    }
                    set_variable = { WFL_staff_dante_grossy_phase = 2 }
                    set_variable = { WFL_staff_dante_grossy_days = WFL_staff_dante_grossy_duration }
                    set_variable = { WFL_staff_dante_grossy_progress = 0 }
                    activate_shine_on_focus = WFL_staff_dante_grossy_2
                }

                else_if = {
                    limit = {
                        check_variable = { var = WFL_staff_dante_grossy_rank value = 2 compare = equals }
                    }
                    set_variable = { WFL_staff_dante_grossy_phase = 2 }
                    set_variable = { WFL_staff_dante_grossy_days = WFL_staff_dante_grossy_duration }
                    set_variable = { WFL_staff_dante_grossy_progress = 0 }
                    activate_shine_on_focus = WFL_staff_dante_grossy_3
                }
            }
        }
        else_if = {
            limit = {
                check_variable = { var = WFL_staff_dante_grossy_phase value = 2 compare = equals }
            }
            add_to_variable = { WFL_staff_dante_grossy_days = -1 }
            set_variable = { WFL_staff_dante_grossy_progress = WFL_staff_dante_grossy_duration }
            subtract_from_variable = { WFL_staff_dante_grossy_progress = WFL_staff_dante_grossy_days }
            divide_variable = { WFL_staff_dante_grossy_progress = WFL_staff_dante_grossy_duration }
            clamp_variable = { var = WFL_staff_dante_grossy_progress min = 0 max = 1 }
            if = {
                limit = {
                    check_variable = { var = WFL_staff_dante_grossy_days value = 0 compare = less_than_or_equals }
                }
                if = {
                    limit = {
                        check_variable = { var = WFL_staff_dante_grossy_rank value = 0 compare = equals }
                    }
                    complete_national_focus = { focus = WFL_staff_dante_grossy_1 use_side_message = yes originator_name = WFL_staff_dante_grossy_name }
                    # Idempotent fallback if an external script already marked this focus complete.
                    WFL_staff_finish_dante_grossy_1 = yes
                }

                else_if = {
                    limit = {
                        check_variable = { var = WFL_staff_dante_grossy_rank value = 1 compare = equals }
                    }
                    complete_national_focus = { focus = WFL_staff_dante_grossy_2 use_side_message = yes originator_name = WFL_staff_dante_grossy_name }
                    # Idempotent fallback if an external script already marked this focus complete.
                    WFL_staff_finish_dante_grossy_2 = yes
                }

                else_if = {
                    limit = {
                        check_variable = { var = WFL_staff_dante_grossy_rank value = 2 compare = equals }
                    }
                    complete_national_focus = { focus = WFL_staff_dante_grossy_3 use_side_message = yes originator_name = WFL_staff_dante_grossy_name }
                    # Idempotent fallback if an external script already marked this focus complete.
                    WFL_staff_finish_dante_grossy_3 = yes
                }
            }
        }
    }
}

            limit = { check_variable = { var = WFL_staff_nominee value = 4 compare = equals } }
            uncomplete_national_focus = { focus = WFL_staff_appoint_max_mueller uncomplete_children = no refund_political_power = no }
        }
        else_if = {
            limit = { check_variable = { var = WFL_staff_nominee value = 5 compare = equals } }

        }
        set_variable = { WFL_staff_nominee = 0 }
    }
    WFL_staff_remove_dante_grossy = yes
    WFL_staff_remove_neal_foerster = yes
    WFL_staff_remove_max_heinsch = yes
    WFL_staff_remove_max_mueller = yes
    WFL_staff_remove_till_hoehne = yes

    # This prevents an old open dialog from acting on a different, newer nomination.
}
