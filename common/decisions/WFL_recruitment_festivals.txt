                        any_owned_state = { NOT = { is_core_of = ROOT } }
                        any_controlled_state = { NOT = { is_core_of = ROOT } }
                    }
                }
            }
        }
        cancel_effect = {
            clr_country_flag = WFL_koenigliche_freibieroffensive_active
            set_country_flag = WFL_koenigliche_freibieroffensive_failed
        }
        remove_effect = {
            if = {
                limit = {
                    has_country_flag = WFL_koenigliche_freibieroffensive_active
                    tag = WFL
                    exists = yes
                    has_capitulated = no
                    has_country_leader = { character = WFL_schuetzenkoenig ruling_only = yes }
