# Campaign owner flags bind the 90-day action to its original host.
# every_country cleanup runs only on campaign completion/cancellation.
# Two paid request decisions per state, each fire_only_once. Second-chain flags
# are separate so delayed first-chain events never resolve the second request.
WFL_schuetzenreferenden = {
    WFL_schuetzenkampagne_51 = {
        visible = {
            OR = {
                NOT = { has_country_flag = WFL_ref_51_campaign_started }
                has_country_flag = WFL_ref_51_campaign_active
            }
        }
        available = {
            exists = yes
            has_war = no
            is_subject = no
            has_capitulated = no
            51 = {
                NOT = { is_owned_by = WFL }
                NOT = { is_core_of = WFL }
                owner = {
                    exists = yes
                    has_war = no
                    is_subject = no
                    has_capitulated = no
                    has_full_control_of_state = 51
                }
            }
        }
        cost = 75
        fire_only_once = yes
        days_remove = 90
        ai_will_do = { base = 1 }
        complete_effect = {
            set_country_flag = WFL_ref_51_campaign_started
            set_country_flag = WFL_ref_51_campaign_active
            51 = { owner = { set_country_flag = WFL_ref_51_campaign_host } }
        }
        cancel_trigger = {
            NOT = {
                exists = yes
                has_war = no
                is_subject = no
                has_capitulated = no
                51 = {
                    NOT = { is_owned_by = WFL }
                    NOT = { is_core_of = WFL }
                    owner = {
                        exists = yes
                        has_war = no
                        is_subject = no
                        has_capitulated = no
                        has_full_control_of_state = 51
                        has_country_flag = WFL_ref_51_campaign_host
                    }
                }
            }
        }
        cancel_effect = {
            clr_country_flag = WFL_ref_51_campaign_active
            set_country_flag = WFL_ref_51_campaign_failed
            every_country = {
                limit = { has_country_flag = WFL_ref_51_campaign_host }
                clr_country_flag = WFL_ref_51_campaign_host
            }
            country_event = { id = wfl_referendum.5 hours = 1 }
        }
        remove_effect = {
            if = {
                limit = { has_country_flag = WFL_ref_51_campaign_active }
                clr_country_flag = WFL_ref_51_campaign_active
                if = {
                    limit = {
                        exists = yes
                        has_war = no
                        is_subject = no
                        has_capitulated = no
                        51 = {
                            NOT = { is_owned_by = WFL }
                            NOT = { is_core_of = WFL }
                            owner = {
                                exists = yes
                                has_war = no
                                is_subject = no
                                has_capitulated = no
                                has_full_control_of_state = 51
                                has_country_flag = WFL_ref_51_campaign_host
                            }
                        }
                    }
                    set_country_flag = WFL_ref_51_campaign_done
                    country_event = { id = wfl_referendum.7 hours = 1 }
                }
                else = {
                    set_country_flag = WFL_ref_51_campaign_failed
                    country_event = { id = wfl_referendum.5 hours = 1 }
                }
            }
            every_country = {
                limit = { has_country_flag = WFL_ref_51_campaign_host }
                clr_country_flag = WFL_ref_51_campaign_host
            }
        }
    }
