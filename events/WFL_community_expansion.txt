# Fictional local comedy. Focuses schedule these once; an options ledger also
# prevents a repeated option/effect from granting the reward a second time.
# Keep leadership/ownership out of the event trigger so changed circumstances
# resolve with compensation instead of silently swallowing a completed focus.
add_namespace = wfl_community

country_event = {
    id = wfl_community.1
    title = wfl_community.1.t
    desc = wfl_community.1.d
    picture = GFX_report_event_generic_parliament
    is_triggered_only = yes
    fire_only_once = yes
    trigger = {
        tag = WFL
        has_completed_focus = WFL_brauereiverbund
        NOT = { has_country_flag = WFL_community_1_resolved }
    }
    option = {
        name = wfl_community.1.a
        ai_chance = { factor = 50 }
        if = {
            limit = { NOT = { has_country_flag = WFL_community_1_resolved } }
            set_country_flag = WFL_community_1_resolved
            if = {
                limit = { has_government = neutrality has_country_leader = { character = WFL_schuetzenkoenig ruling_only = yes } }
                add_stability = 0.05
                add_political_power = 50
            }
            else = { add_political_power = 50 }
        }
    }
    option = {
        name = wfl_community.1.b
        ai_chance = { factor = 50 }
        if = {
            limit = { NOT = { has_country_flag = WFL_community_1_resolved } }
            set_country_flag = WFL_community_1_resolved
            if = {
                limit = { has_government = neutrality has_country_leader = { character = WFL_schuetzenkoenig ruling_only = yes } }
                add_ideas = WFL_festkasse_idee
            }
            else = { add_political_power = 50 }
        }
    }
}
