add_namespace = wfl_beer

country_event = {
    id = wfl_beer.1
    title = wfl_beer.1.t
    desc = wfl_beer.1.d
    picture = GFX_report_event_generic_parliament
    is_triggered_only = yes
    fire_only_once = yes
    trigger = { tag = WFL has_completed_focus = WFL_rheinblockade NOT = { has_country_flag = WFL_beer_rhine_started } }
    option = {
        name = wfl_beer.1.a
        if = {
            limit = { NOT = { has_country_flag = WFL_beer_rhine_started } }
            set_country_flag = WFL_beer_rhine_started
            if = {
                limit = { GER = { exists = yes has_capitulated = no NOT = { has_war_with = WFL } } }
                set_country_flag = WFL_beer_ger_pending
                GER = { country_event = { id = wfl_beer.2 days = 1 } }
            }
            else = { set_country_flag = WFL_beer_ger_resolved add_political_power = 50 }
            if = {
                limit = { HOL = { exists = yes has_capitulated = no NOT = { has_war_with = WFL } } }
                set_country_flag = WFL_beer_hol_pending
                HOL = { country_event = { id = wfl_beer.3 days = 1 } }
            }
            else = { set_country_flag = WFL_beer_hol_resolved add_political_power = 50 }
            country_event = { id = wfl_beer.7 days = 30 }
        }
    }
}
